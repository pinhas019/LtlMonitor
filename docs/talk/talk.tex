% --- THE PROBLEM ---
\begin{frame}{The problem}
    \begin{columns}[c]
        \begin{column}{0.50\textwidth}
            \begin{itemize}
                \item Robots often fail \textbf{in the middle} of a task.
                \item They often \textbf{do not notice}.
                \item They report ``done'' anyway.
                \item \key{Goal:} notice the failure \key{the moment it happens}.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \centering
            \includegraphics[width=\linewidth]{figures/robot_false_success.png}
            \pic{The robot says ``done'' at the drop-off (blue). The box (yellow) is still on the floor.}
        \end{column}
    \end{columns}
    \note{
    \begin{itemize}
        \item The picture is a real run from our simulator: the box slipped while carrying; the robot walked on and reported success.
        \item The robot's own program only knows what it \emph{commanded}, not what really happened.
        \item Finding out at the end is late: the time is spent, and some failures cannot be undone.
        \item So we want an independent watcher that sees the failure when it happens.
    \end{itemize}}
\end{frame}

\begin{frame}{Related work 1: monitors}
    \scriptsize
    \renewcommand{\arraystretch}{1.3}
    \begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.52\linewidth} >{\raggedright\arraybackslash}p{0.44\linewidth}@{}}
        \textbf{\small Work} & \textbf{\small What it does / \color{bad}what is missing} \\ \hline
        \paper{https://doi.org/10.1145/2000799.2000800}{Runtime Verification for LTL and TLTL}{Bauer, Leucker, Schallhart}{2011}
            & Checks LTL rules while the system runs: yes / no / not yet.\newline
              {\color{bad}No task stages: ``eventually'' stays ``not yet'' to the end.} \\
        \paper{https://doi.org/10.1007/978-3-030-63486-5_40}{ROSMonitoring: A Runtime Verification Framework for ROS}{Ferrando et al.}{2020}\newline
        \paper{https://doi.org/10.1007/s10009-023-00720-3}{RTAMT: Runtime Robustness Monitors with Application to CPS and Robotics}{Yamaguchi, Hoxha, Ničković}{2024}
            & Monitors that run next to a robot and read its messages.\newline
              {\color{bad}Rules written by hand; no idea of task progress.} \\
        \paper{https://doi.org/10.1007/978-3-642-29860-8_15}{Runtime Verification with State Estimation}{Stoller et al.}{2011}
            & Guesses a hidden state with a probability model.\newline
              {\color{bad}Needs that model; no task stages.} \\ \hline
    \end{tabular}
    \note{
    \begin{itemize}
        \item \textbf{Bauer et al. 2011 (TOSEM):} the standard theory of LTL monitoring (``LTL$_3$''): after each step, true, false or ``?''; the smallest monitor that answers as early as possible. We build on it. Gap: ``eventually deliver'' stays ``?'' to the end; no stages.
        \item \textbf{ROSMonitoring 2020 (TAROS):} turns ROS topics into monitor nodes; tested on a simulated Mars rover. \textbf{RTAMT 2024 (STTT):} Signal Temporal Logic monitors that say \emph{how much} a rule holds; used with ROS on a Toyota robot. Gap: rules written by hand; no task progress.
        \item \textbf{Stoller et al. 2011 (RV):} when events are missed, a Hidden Markov Model fills the gaps and gives the probability that a rule holds. Closest to our POMDP view. Gap: needs a probability model; no stages.
    \end{itemize}}
\end{frame}

    \vspace{0.05cm}

    \small
    \key{Our gap:} rules made from plain words \textbf{+} task stages \textbf{+} only the robot's own sensors.
    \note{
    \begin{itemize}
        \item \textbf{Lang2LTL (CoRL 2023):} English route commands to LTL (GPT-4 finds places, a small model writes the formula); works in new cities. For planning.
        \item \textbf{nl2spec (CAV 2023):} an LLM translates a sentence to LTL and shows which words became which part, so a person can fix it: 16/36 hard cases alone, 31/36 with a human. For writing specs.

        \item Gap: none makes the monitor from the task description, with stages, from the robot's own sensors.
    \end{itemize}}
\end{frame}

% --- BACKGROUND ---
\begin{frame}{Background: what is a monitor?}
    \begin{columns}[c]
        \begin{column}{0.56\textwidth}
            \begin{itemize}
                \item A program that \textbf{watches} the robot's messages.
                \item It checks \textbf{rules over time}, written in \key{LTL} (a logic of time).
                \item Example: ``\emph{always} stay upright''.
                \item Each rule becomes a small \textbf{state machine}.
                \item Problem: ``\emph{eventually} deliver'' \textbf{never fails early}.
            \end{itemize}
        \end{column}
        \begin{column}{0.42\textwidth}
            \centering
            \includegraphics[width=\linewidth]{figures/aut_fell_over.png}
            \pic{``Always upright'' as a state machine. One fall moves it to state 1 (red), forever.}
        \end{column}
    \end{columns}
    \note{
    \begin{itemize}
        \item LTL = Linear Temporal Logic: rules with ``always'' (G) and ``eventually'' (F).
        \item The tool Spot turns each rule into a small automaton. Every second the monitor feeds it the facts it measured (upright? holding the box?) and takes one step.
        \item ``Always upright'' fails the moment the robot falls: easy.
        \item ``Eventually deliver'' can never fail early, because delivery might still happen later. This is the weakness we fix next.
    \end{itemize}}
\end{frame}

% --- OUR IDEA ---
\begin{frame}{Our idea: the stages of the task}
    \textbf{Each stage has its own rules and its own time limit.}
    \vspace{0.3cm}

    \centering
    \resizebox{0.95\linewidth}{!}{
    \begin{tikzpicture}[
        st/.style={draw=mut, rounded corners=4pt, fill=soft, minimum width=2.4cm, minimum height=1.3cm, align=center},
        hot/.style={st, draw=acc, line width=1.4pt, fill=acc!10},
        arr/.style={-{Stealth[length=2.5mm]}, thick, mut},
        flt/.style={-{Stealth[length=2.2mm]}, dashed, draw=bad}]
        \node[st] (w) {\textbf{Walk to box}\\ $\le 45$~s};
        \node[st, right=0.5cm of w] (g) {\textbf{Grasp}\\ $\le 10$~s};
        \node[hot, right=0.5cm of g] (c) {\textbf{Carry}\\ $\le 60$~s\\ \color{acc}must hold box};
        \node[st, right=0.5cm of c] (p) {\textbf{Place}\\ $\le 10$~s};
        \node[st, right=0.5cm of p, fill=green!8] (d) {\textbf{Done}};
        \node[st, below=0.8cm of c, fill=bad!12, draw=bad, minimum width=9cm] (f) {\color{bad}\textbf{Failure:} a rule broke, too slow, or stuck};
        \draw[arr] (w) -- (g); \draw[arr] (g) -- (c); \draw[arr] (c) -- (p); \draw[arr] (p) -- (d);
        \foreach \n in {w, g, c, p} \draw[flt] (\n.south) -- (f.north -| \n.south);
    \end{tikzpicture}
    }
    \vspace{0.2cm}

    \raggedright
    While carrying, the robot \textbf{must hold the box}. A drop is caught \key{at once}.
    \note{
    \begin{itemize}
        \item We call these stages ``progress states''. Each stage says: what must stay true, what ends the stage, a time limit, and that the robot must keep making progress.
        \item These stages were written by the LLM from the plain description of ``fetch and deliver''.
        \item Key example: in Carry, ``holding the box'' must stay true. A drop breaks it in the next second.
        \item The same fact (``not holding'') is normal while walking to the box. A rule without stages cannot say ``must hold'', because it would be false at the start.
    \end{itemize}}
\end{frame}

% --- THE MATHEMATICAL MODEL ---
\begin{frame}{The mathematical model: a POMDP}
    \begin{columns}[c]
        \begin{column}{0.58\textwidth}
            \small
            \begin{itemize}
                \item \textbf{Hidden} $s$: what really happens (stage, box held?).
                \item \textbf{Seen} $o$: the robot's messages.
                \item \textbf{Belief} $b$: our best guess of $s$.
                \item \textbf{Action} $a$: continue, stop, or re-plan.
            \end{itemize}
            \vspace{-0.1cm}
            \[
                b'(s') \;\propto\; \underbrace{Z(o \mid s')}_{\text{does $o$ fit $s'$?}} \sum_{s} \underbrace{T(s' \mid s, a)}_{\text{how $s$ changes}}\, b(s)
            \]
            \vspace{-0.2cm}

            \key{Today:} the guess is one point (stage + rule states); the update is one rule step.
        \end{column}
        \begin{column}{0.40\textwidth}
            \centering
            \resizebox{\linewidth}{!}{
            \begin{tikzpicture}[
                bx/.style={draw=mut, rounded corners=4pt, fill=soft, minimum height=1.2cm, text width=7.2cm, align=center},
                arr/.style={-{Stealth[length=2.5mm]}, thick, mut}]
                \node[bx, fill=bad!8, draw=bad] (s) {\textbf{Hidden $s$}\\ true stage, box held?};
                \node[bx, below=0.8cm of s] (o) {\textbf{Seen $o$}\\ noisy robot messages};
                \node[bx, below=0.8cm of o, draw=acc, line width=1.4pt, fill=acc!10] (b) {\textbf{Belief $b$}\\ stage + rule states};
                \node[bx, below=0.8cm of b] (a) {\textbf{Action $a$}\\ continue / stop / re-plan};
                \draw[arr] (s) -- node[right, color=mut]{$Z$} (o);
                \draw[arr] (o) -- node[right, color=mut]{update} (b);
                \draw[arr] (b) -- node[right, color=mut]{decide} (a);
                \draw[arr] (a.west) -- ++(-0.7, 0) |- node[pos=0.25, left, color=mut]{$T$} (s.west);
            \end{tikzpicture}
            }
            \pic{\paper{https://doi.org/10.1016/S0004-3702(98)00023-X}{Planning and Acting in Partially Observable Stochastic Domains}{Kaelbling, Littman, Cassandra}{1998}}
        \end{column}
    \end{columns}
    \note{
    \begin{itemize}
        \item POMDP (Kaelbling et al., AI Journal 1998): a hidden state, noisy observations, a belief, and actions chosen from the belief.
        \item The formula, right to left: move the old guess forward with how the world changes ($T$), then weigh each state by how well it explains what we saw ($Z$).
        \item Why stages help: the stage is part of $s$. ``Not holding'' fits ``walking'' but not ``carrying'', so in Carry one observation shows the failure.
        \item Honest: today the guess is one point, the update one automaton step, the action a fixed rule. No probabilities yet (last step of ``To be done'').
        \item Value of information: early knowledge helps only if it changes the outcome; tested with deadlines in Experiment 2.
    \end{itemize}}
\end{frame}

% --- METHOD ---
\begin{frame}{Method: how the monitor is built and run}
    \centering
    \resizebox{\linewidth}{!}{
    \begin{tikzpicture}[
        bx/.style={draw=mut, rounded corners=4pt, fill=soft, minimum height=1.6cm, text width=3.8cm, align=center},
        arr/.style={-{Stealth[length=2.5mm]}, thick, mut}]
        \node[bx] (t) {\textbf{Task in words}\\ ``walk to the box, pick it up''};
        \node[bx, right=0.5cm of t] (l) {\textbf{An LLM}\\ writes the stages and rules};
        \node[bx, right=0.5cm of l] (s) {\textbf{Spot} (a tool)\\ rules $\to$ state machines};
        \node[bx, right=0.5cm of s, draw=acc, line width=1.4pt, fill=acc!10] (m) {\textbf{Monitor}\\ checks once a second};
        \node[bx, right=0.5cm of m, fill=bad!10, draw=bad] (a) {\textbf{Alarm}\\ report, stop or re-plan};
        \node[bx, below=0.9cm of m, text width=8cm] (r) {\textbf{Robot messages} (ROS~2)\\ position, laser, box position};
        \draw[arr] (t) -- (l); \draw[arr] (l) -- (s); \draw[arr] (s) -- (m); \draw[arr] (m) -- (a); \draw[arr] (r) -- (m);
    \end{tikzpicture}
    }
    \vspace{0.1cm}

    \raggedright
    \begin{itemize}
        \item The rules are written \textbf{automatically}, from the task in words.
        \item The monitor uses \textbf{only sensors}. It never asks the planner.
    \end{itemize}
    \note{
    \begin{itemize}
        \item The LLM (Claude) wrote the whole spec in one try; automatic checks made sure it loads and that every fact it uses can be measured. It was saved before any experiment ran.
        \item An ``evaluator'' turns the raw messages into true/false facts each second, e.g.\ ``holding the box'' is inferred from where the box is relative to the robot.
        \item Evaluator and monitor are separate ROS~2 programs, exactly as on a real robot.
        \item The monitor never reads the planner's own status: a test enforces this. (Experiment 1 shows why.)
    \end{itemize}}
\end{frame}

% --- EXPERIMENT 1 ---
\begin{frame}{Experiment 1: a short walk (1800 runs)}
    \begin{columns}[c]
        \begin{column}{0.44\textwidth}
            \begin{itemize}
                \item Stages helped \textbf{little} here: 15 vs.\ 14 of 15 stalls caught.
                \item What mattered: measure progress \key{from the robot's sensors}.
                \item Without that: only \textbf{1 of 15} caught.
            \end{itemize}
        \end{column}
        \begin{column}{0.54\textwidth}
            \centering
            \includegraphics[width=\linewidth]{figures/stall.png}
            \pic{The planner still says ``go''. The robot has stopped. The monitor sees no progress.}
        \end{column}
    \end{columns}
    \note{
    \begin{itemize}
        \item 10 failure types $\times$ 20 seeds $\times$ 3 versions $\times$ 3 reactions = 1800 runs.
        \item Walking has two stages that check the same things, so stages add little: falls are caught in 1--7~s by every version.
        \item Stalls are the difference. ``No progress'' is measured from odometry against the goal; without it only 1 of 15 stalls is caught.
        \item The figure: the planner keeps commanding 0.5~m/s; the robot has stopped. A monitor that trusted the planner would see nothing wrong.
    \end{itemize}}
\end{frame}

% --- EXPERIMENT 2 ---
\begin{frame}{Experiment 2: fetch and deliver (3360 runs)}
    \begin{columns}[c]
        \begin{column}{0.42\textwidth}
            \begin{itemize}
                \item Four stages.
                \item Box dropped: caught after \key{1~s} with stages, 14~s without.
                \item Grasp failed: \key{4~s} vs.\ 19~s.
                \item Earlier in \textbf{all 20} runs.
            \end{itemize}
        \end{column}
        \begin{column}{0.56\textwidth}
            \centering
            \includegraphics[width=\linewidth]{figures/chart_latency.png}
            \pic{Middle value (median) of 20 runs for each failure.}
        \end{column}
    \end{columns}
    \note{
    \begin{itemize}
        \item Walk to a box, grasp it, carry it 2.5--4~m, place it. Evaluator and monitor ran as separate ROS~2 programs; the box position came from a noisy detector (2~cm noise, 5\% missed).
        \item Drop: caught 1~s after it happens, because Carry requires holding. Without stages the monitor only notices when the robot claims success, 14~s later.
        \item Failed grasp: caught by the Grasp stage's time limit (4~s vs.\ 19~s).
        \item Paired by scenario: earlier in 20 of 20 runs, later in none (sign test $p = 1.9\cdot10^{-6}$).
    \end{itemize}}
\end{frame}

% --- DEADLINE ---
\begin{frame}{Does catching early matter?}
    \begin{columns}[c]
        \begin{column}{0.40\textwidth}
            \begin{itemize}
                \item With no time limit: all monitors $\approx$ 82--84\%.
                \item With a time limit: \key{64\%} with stages, 36\% without.
                \item Hand-written checks: 81\%, but someone must \textbf{predict every failure}.
            \end{itemize}
        \end{column}
        \begin{column}{0.58\textwidth}
            \centering
            \includegraphics[width=\linewidth]{figures/chart_deadline.png}
            \pic{Tasks finished on time, 160 runs per bar. Early knowledge pays when time matters.}
        \end{column}
    \end{columns}
    \note{
    \begin{itemize}
        \item On an alarm the robot re-plans. With no deadline, every monitor ends at 82--84\% success, because a late re-plan still works.
        \item With a deadline of 1.5$\times$ the normal time: 64\% with stages vs.\ 36\% without ($p = 7\cdot10^{-7}$). LTL alone (no stages): 36\%.
        \item This is the ``value of information'' from the POMDP slide: early knowledge pays only when time matters.
        \item Hand-written checks inside the robot program do better (81\%), because the program knows its own state. But an engineer had to foresee each failure: see the next slide.
    \end{itemize}}
\end{frame}

% --- UNPLANNED FAILURES ---
\begin{frame}{Failures nobody planned for}
    \begin{columns}[c]
        \begin{column}{0.48\textwidth}
            \begin{itemize}
                \item Box left at the wrong place; robot too close to a wall.
                \item Hand-written checks: \textbf{\color{bad}40 of 40} missed.
                \item With our monitor: \key{0 of 40} missed.
                \item Missed by all: too slow, or a long detour.
            \end{itemize}
        \end{column}
        \begin{column}{0.50\textwidth}
            \centering
            \includegraphics[width=\linewidth]{figures/robot_wrong_dropoff.png}
            \pic{The box is far from the drop-off (blue), yet the robot says ``done''. Our monitor catches it.}
        \end{column}
    \end{columns}
    \note{
    \begin{itemize}
        \item New failures, one per requirement of the task description that the hand-written checks do not cover. The design and predictions were written down before the run.
        \item Wrong drop-off: the robot's own idea of the drop-off is wrong, so its own check passes and it reports success. Too close: the navigator grazes obstacles.
        \item Hand-written checks alone: 40 of 40 failed silently. Adding our monitor: 0 of 40, at no measurable cost elsewhere.
        \item Missed by everyone: running slowly, or a detour, that stay inside the time limits. Our prediction for ``slow'' was wrong, and we say so.
    \end{itemize}}
\end{frame}

\begin{frame}{What we learned}
    \begin{columns}[c]
        \begin{column}{0.50\textwidth}
            \begin{itemize}
                \item \key{Stages catch failures when they happen.}
                \item Early alarms help \textbf{when time matters}.
                \item Rules come from words, so failures \textbf{nobody planned for} are caught too.
                \item Use the \textbf{robot's own sensors}, not the planner.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \centering
            \includegraphics[width=\linewidth]{figures/demo_page.png}
            \pic{Live demo, one command:\\ \texttt{docker run -p 8080:8080 ghcr.io/pinhas019/ltlmonitor-demo}}
        \end{column}
    \end{columns}
    \note{
    \begin{itemize}
        \item Four take-aways, one per line on the slide.
        \item If there is time, show the demo: open http://localhost:8080, click ``Dropped object: full spec'', press Pause at 22~s.
        \item The demo page draws the stages and the automaton live, from the monitor's own ROS~2 messages.
    \end{itemize}}
\end{frame}
