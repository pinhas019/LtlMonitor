% --- 2: THE PROBLEM ---
\begin{frame}{The problem: robots fail silently}
    \begin{columns}[c]
        \begin{column}{0.52\textwidth}
            \begin{itemize}
                \item A task has steps: walk, pick up, carry, put down.
                \item Things go wrong: the box slips, the grasp misses.
                \item Often the robot \textbf{does not know}, and reports ``done''.
                \item We want an \key{independent watcher} that notices \key{at once}.
            \end{itemize}
        \end{column}
        \begin{column}{0.46\textwidth}
            \centering
            \includegraphics[width=\linewidth]{figures/robot_false_success.png}
            \pic{The robot reports success at the drop-off (blue). The box (yellow) is still on the floor.}
        \end{column}
    \end{columns}
\end{frame}

% --- 3: BACKGROUND ---
\begin{frame}{Background: a runtime monitor}
    \begin{columns}[c]
        \begin{column}{0.56\textwidth}
            \begin{itemize}
                \item A \textbf{separate program} that only reads the robot's messages.
                \item It checks \textbf{rules over time}, in \key{LTL}: ``\emph{always} upright'', ``\emph{eventually} deliver''.
                \item Each rule becomes a small \textbf{automaton}; one step per second.
                \item Catch: ``eventually'' \textbf{never fails early}.
            \end{itemize}
        \end{column}
        \begin{column}{0.42\textwidth}
            \centering
            \includegraphics[width=\linewidth]{figures/aut_fell_over.png}
            \pic{``Always upright'': one fall moves to state 1 (red) for good.}
        \end{column}
    \end{columns}
\end{frame}

% --- 5: THE MODEL (DEFINITION) ---
\begin{frame}{The model behind it: a POMDP}
    \begin{columns}[c]
        \begin{column}{0.58\textwidth}
            \textbf{Definition:} a POMDP $\langle S, A, O, T, Z, R\rangle$; the agent keeps a belief $b$ over hidden states:
            \[
                b'(s') \;\propto\; Z(o \mid s') \sum_{s} T(s' \mid s, a)\, b(s)
            \]
            \vspace{-0.4cm}
            {\small
            \begin{itemize}
                \item $s$ includes the \key{true stage}; $o$ = ROS~2 messages; $a$ = continue / stop / re-plan.
                \item So one $o$ means different things: ``not holding'' is fine while walking, a fault while carrying.
            \end{itemize}}
        \end{column}
        \begin{column}{0.40\textwidth}
            \centering
            \resizebox{\linewidth}{!}{
            \begin{tikzpicture}[
                bx/.style={draw=mut, rounded corners=4pt, fill=soft, minimum height=1.2cm, text width=7.2cm, align=center},
                arr/.style={-{Stealth[length=2.5mm]}, thick, mut}]
                \node[bx, fill=bad!8, draw=bad] (s) {\textbf{Hidden state $s$}\\ true stage, box held?};
                \node[bx, below=0.8cm of s] (o) {\textbf{Observation $o$}\\ noisy ROS~2 messages};
                \node[bx, below=0.8cm of o, draw=acc, line width=1.4pt, fill=acc!10] (b) {\textbf{Belief $b$}\\ stage + automaton states};
                \node[bx, below=0.8cm of b] (a) {\textbf{Action $a$}\\ continue / stop / re-plan};
                \draw[arr] (s) -- node[right, color=mut]{$Z$} (o);
                \draw[arr] (o) -- node[right, color=mut]{update} (b);
                \draw[arr] (b) -- node[right, color=mut]{policy} (a);
                \draw[arr] (a.west) -- ++(-0.7, 0) |- node[pos=0.25, left, color=mut]{$T$} (s.west);
            \end{tikzpicture}
            }
        \end{column}
    \end{columns}
\end{frame}

% --- 6: WHAT WE COMPUTE (THEOREM / DEFINITION TYPE) ---
\begin{frame}{What our monitor computes today}
    \textbf{Claim:} the monitor is a \key{deterministic belief filter} with a fixed policy.

    \begin{enumerate}
        \item Belief = one point: $b = (\text{stage}, q_1, \dots, q_k)$; $q_i$ = state of automaton $i$.
        \item Update = one automaton step: $q_i' = \delta_i(q_i, \sigma(o))$.
        \item Policy = fixed: a broken rule $\Rightarrow$ report, stop or re-plan.
    \end{enumerate}
    \vspace{0.2cm}
    \textbf{Not yet:} probabilities and an optimised policy. That is the next step.
    \vspace{0.1cm}

    {\small\color{mut} $\sigma(o)$: true/false propositions computed from the messages, e.g.\ \texttt{holding\_object}.}
\end{frame}

\begin{frame}{Method: how the monitor is built and run}
    \centering
    \resizebox{\linewidth}{!}{

    \end{itemize}
\end{frame}

% --- 9: EXPERIMENT 1 ---
\begin{frame}{Experiment 1: a short walk (1800 runs)}
    \begin{columns}[c]
        \begin{column}{0.44\textwidth}
            \begin{itemize}
                \item Two similar stages: they \textbf{added little} (15 vs.\ 14 of 15 stalls).
                \item What mattered: \key{progress from the robot's own sensors}. Without it: \textbf{1 of 15}.
            \end{itemize}
            \textbf{Lesson:} don't trust the planner's report.
        \end{column}
        \begin{column}{0.54\textwidth}
            \centering
            \includegraphics[width=\linewidth]{figures/stall.png}
            \pic{The planner commands 0.5 m/s; the robot has stopped. The monitor sees no progress.}
        \end{column}
    \end{columns}
\end{frame}

% --- 10: EXPERIMENT 2 ---

    \begin{columns}[c]

% --- 12: DEADLINE ---
\begin{frame}{Does early matter? Yes, with a deadline}
    \begin{columns}[c]
        \begin{column}{0.40\textwidth}
            \begin{itemize}
                \item No deadline: all monitors $\approx$ 82--84\%.
                \item Deadline: \key{64\%} with stages vs.\ 36\%. LTL alone: 36\%.
                \item Hand-written checks: 81\%, but each fault must be \textbf{foreseen}.
            \end{itemize}
        \end{column}
        \begin{column}{0.58\textwidth}
            \centering
            \includegraphics[width=\linewidth]{figures/chart_deadline.png}
            \pic{160 runs per bar. Value of information, as the POMDP view predicts.}
        \end{column}
    \end{columns}
\end{frame}

% --- 14: WHAT WE LEARNED ---
\begin{frame}{What we learned}
    \begin{columns}[c]
        \begin{column}{0.50\textwidth}
            \begin{itemize}
                \item \key{Stages catch failures as they happen.}
                \item Early pays off \textbf{when time matters}.
                \item Generated from words; catches faults \textbf{nobody planned for}.
                \item Progress from the \textbf{robot's own sensors}.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \centering
            \includegraphics[width=\linewidth]{figures/demo_page.png}

    \end{columns}
\end{frame}
